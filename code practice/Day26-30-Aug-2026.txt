#Ansible — Implementing SSH Key-based Authentication
#***************************************************
---
- name: Implement SSH Key Authentication
  hosts: all
  become: true

  tasks:
    - name: Generate SSH key pair
      ansible.builtin.command:
        cmd: ssh-keygen -t rsa -b 2048 -f /home/user/.ssh/id_rsa -N ""
        creates: /home/user/.ssh/id_rsa

    - name: Add SSH public key to authorized_keys
      ansible.posix.authorized_key:
        user: user
        state: present
        key: "{{ lookup('ansible.builtin.file', '/home/user/.ssh/id_rsa.pub') }}"

## CloudWatch Log Group
#********************************************************************************
resource "aws_cloudwatch_log_group" "pathnex_flow_logs_group" {
  name = "pathnex-flow-logs"
}


# IAM Role for VPC Flow Logs
resource "aws_iam_role" "pathnex_flow_log_role" {
  name = "pathnex-flow-log-role"

  assume_role_policy = jsonencode({
    Version = "2012-10-17"

    Statement = [
      {
        Effect = "Allow"

        Principal = {
          Service = "vpc-flow-logs.amazonaws.com"
        }

        Action = "sts:AssumeRole"
      }
    ]
  })
}


# IAM Policy for VPC Flow Logs
resource "aws_iam_role_policy" "pathnex_flow_log_policy" {
  name = "pathnex-flow-log-policy"
  role = aws_iam_role.pathnex_flow_log_role.id

  policy = jsonencode({
    Version = "2012-10-17"

    Statement = [
      {
        Effect = "Allow"

        Action = [
          "logs:CreateLogGroup",
          "logs:CreateLogStream",
          "logs:PutLogEvents",
          "logs:DescribeLogGroups",
          "logs:DescribeLogStreams"
        ]

        Resource = "*"
      }
    ]
  })
}


# VPC Flow Logs
resource "aws_vpc_flow_log" "pathnex_flow_log" {
  vpc_id = aws_vpc.pathnex_vpc.id

  traffic_type = "ALL"

  log_destination_type = "cloud-watch-logs"
  log_destination      = aws_cloudwatch_log_group.pathnex_flow_logs_group.arn

  iam_role_arn = aws_iam_role.pathnex_flow_log_role.arn

  depends_on = [
    aws_iam_role_policy.pathnex_flow_log_policy
  ]
}

#Kubernetes — Network Policies for Security
#***********************************************************************************
apiVersion: networking.k8s.io/v1
kind: NetworkPolicy
metadata:
  name: pathnex-web-policy
spec:
  podSelector:
    matchLabels:
      app: pathnex-web-app

  policyTypes:
    - Ingress
    - Egress

  ingress:
    - from:
        - podSelector:
            matchLabels:
              app: pathnex-api
      ports:
        - protocol: TCP
          port: 80

  egress:
    - to:
        - podSelector:
            matchLabels:
              app: pathnex-database
      ports:
        - protocol: TCP
          port: 3306

#Jenkinsfile — Secure the Pipeline with Secrets
#*********************************************
pipeline {

    agent any

    environment {
        AWS_ACCESS_KEY_ID     = credentials('aws-access-key')
        AWS_SECRET_ACCESS_KEY = credentials('aws-secret-key')
        AWS_DEFAULT_REGION    = 'us-east-1'
    }

    stages {

        stage('Build') {
            steps {
                echo 'Building Docker image...'
                script {
                    docker.build('pathnex-web-app')
                }
            }
        }

        stage('Deploy') {
            steps {
                echo 'Deploying to AWS EC2...'

                sh '''
                    aws ec2 run-instances \
                      --image-id ami-0abcd1234 \
                      --count 1 \
                      --instance-type t2.micro
                '''
            }
        }
    }
}
