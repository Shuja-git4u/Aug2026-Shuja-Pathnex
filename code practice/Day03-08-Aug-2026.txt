#Ansible — Install Apache and Start Service
#**********************************************
- name: Install and Start Apache Service
  hosts: all
  become: yes
  
  tasks:
   - name: Install Apache
     yum:
      name: httpd
      state: present
   
   - name: Start apache service
      name: httpd
      state: started
      enabled: yes
