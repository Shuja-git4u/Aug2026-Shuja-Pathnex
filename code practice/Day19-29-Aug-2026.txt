# Ansible — Setup Prometheus for Monitoring
#***************************************************
---
- name: Install Prometheus for Monitoring
  hosts: all
  become: true

  tasks:
    - name: Install Prometheus and Node Exporter
      ansible.builtin.yum:
        name:
          - prometheus
          - prometheus-node-exporter
        state: present

    - name: Start and enable Prometheus service
      ansible.builtin.service:
        name: prometheus
        state: started
        enabled: true

# Create a CloudWatch Log Group

  name = "pathnex-log-group"

  name           = "pathnex-log-stream"
  log_group_name = aws_cloudwatch_log_group.pathnex_log_group.name

#*****************************************************************
#Kubernetes — Deploy Fluentd for Logging
#*****************************************************************
---
apiVersion: apps/v1
kind: Deployment
metadata:
  name: fluentd
  labels:
    app: fluentd

spec:
  replicas: 1

  selector:
    matchLabels:
      app: fluentd

  template:
    metadata:
      labels:
        app: fluentd

    spec:
      containers:
        - name: fluentd
          image: fluent/fluentd:latest

          ports:
            - containerPort: 24224

---
apiVersion: v1
kind: Service
metadata:
  name: fluentd

spec:
  selector:
    app: fluentd

  ports:
    - protocol: TCP
      port: 24224
      targetPort: 24224

  type: ClusterIP

            }
        }

        stage('Test') {

            }
        }

        stage('Deploy and Monitor') {
            steps {
                script {
                    echo 'Deploying application to Kubernetes...'
                    sh 'kubectl apply -f kubernetes/deployment.yaml'

                    echo 'Checking deployment status...'
