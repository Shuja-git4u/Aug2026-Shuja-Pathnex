#Ansible — Setup Prometheus Node Exporter for Monitoring
#*******************************************************
---
- name: Setup Prometheus Node Exporter
  hosts: all
  become: true

  tasks:
    - name: Install Prometheus Node Exporter
      ansible.builtin.yum:
        name: prometheus-node-exporter
        state: present

    - name: Start and enable Prometheus Node Exporter
      ansible.builtin.service:
        name: prometheus-node-exporter
        state: started
        enabled: true

# Terraform — Create CloudWatch Alarm
#*****************************************************

resource "aws_cloudwatch_metric_alarm" "pathnex_cpu_alarm" {
  alarm_name          = "pathnex-cpu-utilization-high"
  comparison_operator = "GreaterThanThreshold"

  evaluation_periods = 1
  metric_name        = "CPUUtilization"
  namespace          = "AWS/EC2"
  period             = 60
  statistic          = "Average"
  threshold          = 80

  dimensions = {
    InstanceId = aws_instance.pathnex_ec2.id
  }

  alarm_description = "Alarm when EC2 CPU utilization exceeds 80%"

  alarm_actions = [
    aws_sns_topic.pathnex_alerts.arn
  ]

  treat_missing_data = "notBreaching"
}


# SNS Topic

resource "aws_sns_topic" "pathnex_alerts" {
  name = "pathnex-alerts"
}

# Jenkinsfile — Continuous Monitoring and Alerts in CI Pipeline
#**************************************************************
pipeline {
    agent any

    stages {

        stage('Monitor') {
            steps {
                echo 'Publishing CPU metric to CloudWatch...'

                sh '''
                    aws cloudwatch put-metric-data \
                      --namespace "AWS/EC2" \
                      --metric-name "CPUUtilization" \
                      --value 90 \
                      --unit Percent
                '''
            }
        }

        stage('Test') {
            steps {
                echo 'Running tests...'
            }
        }
    }
}
