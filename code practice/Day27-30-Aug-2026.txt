Day 27 — Serverless Scaling with Lambda and DynamoDB
🔹 Ansible — Deploy Serverless Lambda Functions
#****************************************************
---
- name: Deploy Serverless Lambda Function
  hosts: localhost
  gather_facts: false

  tasks:
    - name: Create Lambda function
      ansible.builtin.command:
        cmd: >
          aws lambda create-function
          --function-name pathnex-lambda
          --runtime nodejs22.x
          --role arn:aws:iam::123456789012:role/lambda-role
          --handler index.handler
          --zip-file fileb://lambda.zip
#**********************************************************
#Terraform — Create DynamoDB Table for Serverless Application
#**********************************************************
resource "aws_dynamodb_table" "pathnex_table" {
  name         = "pathnex-table"
  billing_mode = "PAY_PER_REQUEST"
  hash_key     = "id"

  attribute {
    name = "id"
    type = "S"
  }
}
#************************************************************
#Kubernetes — Deploy Serverless Using Kubeless
#************************************************************
apiVersion: kubeless.io/v1beta1
kind: Function
metadata:
  name: pathnex-lambda
  namespace: default

spec:
  runtime: nodejs22
  handler: index.handler

  function: |
    module.exports.handler = function(event, context) {
      return "Hello World";
    };
#*************************************************************
#Jenkinsfile — Lambda Function Deploymen
#*************************************************************
pipeline {
    agent any

    stages {

        stage('Build') {
            steps {
                echo 'Building Lambda function package...'
                sh 'zip -r lambda.zip lambda/'
            }
        }

        stage('Deploy to Lambda') {
            steps {
                echo 'Deploying Lambda function...'
                sh 'aws lambda update-function-code --function-name pathnex-lambda --zip-file fileb://lambda.zip'
            }
        }
    }
}

docker stats

# Run Nginx with resource limits
docker run -d \
  --name pathnex-nginx \
  --memory="512m" \
  --cpus="1.0" \
  nginx

docker exec -it pathnex-nginx bash
