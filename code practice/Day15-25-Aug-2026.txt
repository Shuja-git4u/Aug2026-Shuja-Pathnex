#****************************************************************
#Ansible — Deploy a Web Application to Multiple Hosts
#****************************************************************
---
- name: Deploy web application to multiple hosts
  hosts: all
  become: yes

  tasks:
    - name: Copy web app files
      ansible.builtin.copy:
        src: /pathnex/app/
        dest: /var/www/html/

    - name: Start web app service
      ansible.builtin.service:
        name: apache2
        state: started
        enabled: yes

#Terraform — Create EC2 with ALB (Application Load Balancer)
#****************************************************************
# Application Load Balancer (ALB)
*****************************************************************
resource "aws_lb" "pathnex_lb" {
  name               = "pathnex-lb"
  internal           = false
  load_balancer_type = "application"

  security_groups = [
    aws_security_group.allow_ssh_http.id
  ]

  # ALB requires subnets in at least two Availability Zones
  subnets = [
    aws_subnet.public_1.id,
    aws_subnet.public_2.id
  ]

  tags = {
    Name = "Pathnex-ALB"
  }
}


# EC2 Instance
resource "aws_instance" "pathnex_ec2" {
  ami           = "ami-0abcd1234abcd1234"

  vpc_security_group_ids = [
    aws_security_group.allow_ssh_http.name

}

# Jenkinsfile — Multi-Environment Deployment
#*************************************************************
pipeline {
    agent any

    environment {
        IMAGE_NAME = 'pathnex-web-app'
        DEPLOY_ENV = 'production'
    }

    stages {
        stage('Build') {
            steps {
                script {
                    echo 'Building Docker image...'
                    docker.build("${env.IMAGE_NAME}")
                }
            }
        }

        stage('Test') {
            steps {
                echo 'Running tests...'

                // Add your test commands here
                // sh 'npm test'
            }
        }

        stage('Deploy to Kubernetes') {
            steps {
                script {
                    if (env.DEPLOY_ENV == 'production') {
                        sh 'kubectl apply -f kubernetes/prod-deployment.yaml'
                    } else {
                        sh 'kubectl apply -f kubernetes/dev-deployment.yaml'
                    }
                }
            }
        }
    }
}
#******************************************************************************
#GitLab CI/CD — Environment-Specific Deployments
#******************************************************************************
stages:
  - build
  - push
  - deploy

variables:
  IMAGE_NAME: "$CI_REGISTRY_IMAGE/pathnex-web-app"

build:
  stage: build
  script:
    - docker build -t "$IMAGE_NAME:$CI_COMMIT_SHORT_SHA" .

push:
  stage: push
  script:
    - docker login -u "$CI_REGISTRY_USER" -p "$CI_REGISTRY_PASSWORD" "$CI_REGISTRY"
    - docker push "$IMAGE_NAME:$CI_COMMIT_SHORT_SHA"

deploy:
  stage: deploy
  script:
    - |
      if [ "$CI_COMMIT_REF_NAME" = "main" ]; then
        kubectl apply -f kubernetes/prod-deployment.yaml
      else
        kubectl apply -f kubernetes/dev-deployment.yaml
      fi
