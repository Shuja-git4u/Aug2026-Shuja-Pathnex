#Day 22 — Multi-Cluster Kubernetes Management
#Ansible — Deploy Nginx Across Multiple Kubernetes Clusters
#**********************************************************
---
- name: Deploy Nginx across multiple Kubernetes clusters
  hosts: localhost
  gather_facts: false

  tasks:
    - name: Deploy Nginx on Cluster 1
      ansible.builtin.command:
        cmd: kubectl --context=us-east-cluster apply -f kubernetes/deployment.yaml

    - name: Deploy Nginx on Cluster 2
      ansible.builtin.command:
        cmd: kubectl --context=us-west-cluster apply -f kubernetes/deployment.yaml

#Terraform — Multi-Region Kubernetes Cluster with EKS and GKE
#*********************************************************************************
# AWS EKS Cluster
resource "aws_eks_cluster" "pathnex_eks" {
  name     = "pathnex-eks-cluster"
  role_arn = aws_iam_role.eks_cluster_role.arn

  vpc_config {
    subnet_ids = [
      aws_subnet.pathnex_subnet.id
    ]
  }

  depends_on = [
    aws_iam_role_policy_attachment.eks_cluster_role_policy
  ]
}


# Google GKE Cluster
resource "google_container_cluster" "pathnex_gke" {
  name     = "pathnex-gke-cluster"
  location = "us-central1-a"

  initial_node_count = 3

  node_config {
    machine_type = "n1-standard-1"
  }
}
#For EKS, it is generally recommended to provide at least two subnets in different Availability Zones:
vpc_config {
  subnet_ids = [
    aws_subnet.pathnex_subnet_1.id,
    aws_subnet.pathnex_subnet_2.id
  ]
}

# Jenkinsfile — Multi-Cluster Deployment Strategy
#********************************************
pipeline {
    agent any

    stages {

        stage('Deploy to Cluster 1') {
            steps {
                sh 'kubectl --context=us-east-cluster apply -f kubernetes/deployment.yaml'
            }
        }

        stage('Deploy to Cluster 2') {
            steps {
                sh 'kubectl --context=us-west-cluster apply -f kubernetes/deployment.yaml'
            }
        }
    }
}
