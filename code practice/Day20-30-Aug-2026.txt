# Ansible — Configure Firewall with iptables
#*********************************************
---
- name: Configure Firewall with iptables
  hosts: all
  become: true

  tasks:
    - name: Set default INPUT policy to DROP
      ansible.builtin.command: iptables -P INPUT DROP

    - name: Allow established and related connections
      ansible.builtin.command: iptables -A INPUT -m conntrack --ctstate ESTABLISHED,RELATED -j ACCEPT

    - name: Allow loopback traffic
      ansible.builtin.command: iptables -A INPUT -i lo -j ACCEPT

    - name: Allow SSH traffic
      ansible.builtin.command: iptables -A INPUT -p tcp --dport 22 -j ACCEPT

    - name: Allow HTTP traffic
      ansible.builtin.command: iptables -A INPUT -p tcp --dport 80 -j ACCEPT
#***************************************************************************
#Terraform — Configure Security Groups for EC2
#***************************************************************************
resource "aws_security_group" "pathnex_sg" {
  name_prefix = "pathnex-sg-"
  description = "Allow SSH and HTTP"

  ingress {
    description = "Allow SSH"
    from_port   = 22
    to_port     = 22
    protocol    = "tcp"
    cidr_blocks = ["0.0.0.0/0"]
  }

  ingress {
    description = "Allow HTTP"
    from_port   = 80
    to_port     = 80
    protocol    = "tcp"
    cidr_blocks = ["0.0.0.0/0"]
  }

  egress {
    description = "Allow all outbound traffic"
    from_port   = 0
    to_port     = 0
    protocol    = "-1"
    cidr_blocks = ["0.0.0.0/0"]
  }

  tags = {
    Name = "pathnex-security-group"
  }
}

pipeline {
    agent any

    environment {
        IMAGE_NAME = 'pathnex-web-app'
    }

    stages {

        stage('Build') {
            steps {
                script {
                    echo 'Building Docker image...'
                    docker.build("${env.IMAGE_NAME}")
                }
            }
        }

        stage('Push') {
            steps {
                script {
                    echo 'Pushing Docker image...'

                    docker.withRegistry(
                        'https://index.docker.io/v1/',
                        'docker-credentials'
                    ) {
                        docker.image("${env.IMAGE_NAME}").push('latest')
                    }
                }
            }
        }
    }
}

#GitLab CI/CD — Secure Deployment with Secrets Management
#************************************************************************
stages:
  - build
  - push
  - deploy

variables:
  IMAGE_NAME: "$CI_REGISTRY_IMAGE/pathnex-web-app"
  IMAGE_TAG: "$CI_COMMIT_SHORT_SHA"

build:
  stage: build
  script:
    - docker build -t "$IMAGE_NAME:$IMAGE_TAG" .

push:
  stage: push
  script:
    - echo "$CI_REGISTRY_PASSWORD" | docker login "$CI_REGISTRY" \
        --username "$CI_REGISTRY_USER" \
        --password-stdin
    - docker push "$IMAGE_NAME:$IMAGE_TAG"

deploy:
  stage: deploy
  script:
    - kubectl apply -f kubernetes/deployment.yaml
    - kubectl rollout status deployment/pathnex-web-app
    - kubectl get pods

#Docker Networking
#*************************************************************************
# Create a custom Docker network
docker network create pathnex-network

# Run the first Ubuntu container
docker run -dit \
  --name c1 \
  --network pathnex-network \
  ubuntu

# Run the second Ubuntu container
docker run -dit \
  --name c2 \
  --network pathnex-network \
  ubuntu
#********************************
#Verify the containers
docker ps
#Test connectivity between containers

#Enter the first container:

docker exec -it c1 bash

#Then try to ping c2:

ping c2

#Because both containers are on the same custom Docker network, Docker's internal DNS allows c1 to resolve c2 by its container name.

#Note: The default Ubuntu image may not have the ping command installed. You can install it inside the container if needed:

apt update
apt install -y iputils-ping
