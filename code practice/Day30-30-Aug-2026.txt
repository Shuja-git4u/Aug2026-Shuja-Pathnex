      version = "~> 6.0"
    }

  region = var.aws_region
}

module "pathnex_vpc" {
  source = "./vpc"

  project_name        = var.project_name
  vpc_cidr            = var.vpc_cidr
  public_subnet_cidrs = var.public_subnet_cidrs
  private_subnet_cidrs = var.private_subnet_cidrs
  availability_zones  = var.availability_zones
}

resource "aws_eks_cluster" "pathnex" {
  name     = "${var.project_name}-eks"
  role_arn = aws_iam_role.eks_cluster_role.arn

  vpc_config {
    subnet_ids = module.pathnex_vpc.private_subnet_ids
  }

  depends_on = [
    aws_iam_role_policy_attachment.eks_cluster_policy,
    aws_iam_role_policy_attachment.eks_vpc_resource_controller
  ]

  tags = {
    Name = "${var.project_name}-eks"
  }
}

  cluster_name    = aws_eks_cluster.pathnex.name
  node_group_name = "${var.project_name}-nodes"

  instance_types = [var.eks_node_instance_type]

    desired_size = 2
    min_size     = 1

    aws_iam_role_policy_attachment.eks_worker_node_policy,
    aws_iam_role_policy_attachment.eks_cni_policy,
    aws_iam_role_policy_attachment.eks_container_registry_read_only

resource "aws_subnet" "public" {
  count = length(var.public_subnet_cidrs)

  vpc_id                  = aws_vpc.pathnex.id
  cidr_block              = var.public_subnet_cidrs[count.index]
  availability_zone       = var.availability_zones[count.index]
  map_public_ip_on_launch = true

  tags = {
    Name = "${var.project_name}-public-${count.index + 1}"
  }
}

  cidr_block        = var.private_subnet_cidrs[count.index]

6. Jenkinsfile
jenkins/Jenkinsfile
#******************************************
pipeline {
    agent any

    environment {
        IMAGE_NAME = "pathnex-web-app"
        IMAGE_TAG  = "${BUILD_NUMBER}"
        HELM_RELEASE = "pathnex-app"
        HELM_CHART   = "./helm/pathnex-app"
        K8S_NAMESPACE = "pathnex"
    }

    stages {

        stage('Checkout') {
            steps {
                checkout scm
            }
        }

        stage('Build Docker Image') {
            steps {
                echo 'Building Docker image...'

                sh '''
                    docker build \
                      -t ${IMAGE_NAME}:${IMAGE_TAG} \
                      -f docker/Dockerfile .
                '''
            }
        }

        stage('Test') {
            steps {
                echo 'Validating Kubernetes and Helm configuration...'

                sh '''
                    helm lint ${HELM_CHART}
                    kubectl apply --dry-run=client -f kubernetes/
                '''
            }
        }

        stage('Deploy') {
            steps {
                echo 'Deploying Pathnex application...'

                sh '''
                    helm upgrade --install ${HELM_RELEASE} ${HELM_CHART} \
                      --namespace ${K8S_NAMESPACE} \
                      --create-namespace \
                      --set image.repository=${IMAGE_NAME} \
                      --set image.tag=${IMAGE_TAG}
                '''
            }
        }

        stage('Verify Deployment') {
            steps {
                sh '''
                    kubectl rollout status \
                      deployment/pathnex-app \
                      -n ${K8S_NAMESPACE} \
                      --timeout=180s

                    kubectl get pods -n ${K8S_NAMESPACE}
                    kubectl get svc -n ${K8S_NAMESPACE}
                    kubectl get hpa -n ${K8S_NAMESPACE}
                '''
            }
        }
    }

    post {
        success {
            echo 'Pathnex application deployed successfully!'
        }

        failure {
            echo 'Pathnex deployment failed.'
        }

        always {
            sh 'docker image prune -f || true'
        }
    }
}

    - docker build -t "$IMAGE_NAME:$IMAGE_TAG" -f docker/Dockerfile .
    - docker push "$IMAGE_NAME:$IMAGE_TAG"

  stage: deploy
  image:
    name: alpine/helm:3.18.0

        --namespace pathnex

        --set image.tag="$IMAGE_TAG"

Deployment sequence

Once these files are in GitHub, your basic deployment sequence is:

Provision AWS
cd terraform

terraform init
terraform fmt -recursive
terraform validate
terraform plan
terraform apply
#*******************************
Then configure kubectl for EKS:

aws eks update-kubeconfig \
  --region us-east-1 \
  --name pathnex-eks
#*************************************
Deploy Kubernetes
kubectl apply -f kubernetes/namespace.yaml
kubectl apply -f kubernetes/deployment.yaml
kubectl apply -f kubernetes/service.yaml
kubectl apply -f kubernetes/hpa.yaml
kubectl apply -f kubernetes/network-policy.yaml

Or use Helm:

helm upgrade --install pathnex-app ./helm/pathnex-app \
  --namespace pathnex \
  --create-namespace
#**************************************************************
Verify:

kubectl get pods -n pathnex
kubectl get svc -n pathnex
kubectl get hpa -n pathnex
