🔹 Ansible Task — Install Nginx on Pathnex Server
#---------------------------------------------------
- name: Install Nginx on Pathnex Server
  hosts: all
  become: yes
  tasks:
  - name: Install nginx
    yum:
      name: nginx
      state: present

Git Integration
#---------------------------------------------------
 Jenkins Pipeline — Checkout Git
# *******************************
# How to checkout Git Repository and List files
#---------------------------------------------------

pipeline {
  agent any {

  stages {
    stage('Checkout') {
      steps {
        git branch: 'main',url:'https://github.com/Pathnex/sample-java-app.git'
          }
         }
    stage('List files') {
        steps {
          sh 'ls -la'
        }
       }
      }
    }
#-----------------------------------------------------
GitLab CI — Checkout Git
#-----------------------------------------------------
# How to Checkout Git repo in GitLab CI
#-----------------------------------------------------

stages:
. checkout
git-checkout:
  stage:
    checkout script:
      - git clone https://github.com/Pathnex/sample-java-app.git
      - cd sample-java-app
      - ls -la
